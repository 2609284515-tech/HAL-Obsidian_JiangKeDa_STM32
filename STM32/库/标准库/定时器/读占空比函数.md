//读占空比
uint32_t IC_Getduty(void)
{
	return (TIM_GetCapture2(TIM3) + 1) * 100 / (TIM_GetCapture1(TIM3) + 1);
}

# HAL库

```
uint32_t IC_Getduty(void)
{
	return (HAL_TIM_ReadCapturedValue(&htim3, TIM_CHANNEL_2) + 1) * 100 / (HAL_TIM_ReadCapturedValue(&htim3, TIM_CHANNEL_1) + 1);
}
```
